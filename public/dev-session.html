terminal Tool Handler: app/api/opal/tools/fetch_cms12_content/route.ts

POST /api/opal/tools/fetch_cms12_content

The execution route receives structured arguments from Mark, maps them to CMS 12 Content Delivery API v3.0 with expand=*, and returns structured JSON. 


// app/api/opal/tools/fetch_cms12_content/route.ts
export async function POST(request: NextRequest) {
  const body = await request.json();
  const args = body.parameters && typeof body.parameters === 'object' ? body.parameters : body;

  const contentId = str(args.content_id);
  const matchUrl = str(args.match_url);
  const expand = str(args.expand) ?? '*';

  // Construct standard CMS 12 Content Delivery REST endpoint
  let path = '/api/episerver/v3.0/content';
  if (contentId) path += '/' + contentId + (args.sub_path ? '/' + args.sub_path : '');

  const res = executeCmsEndpoint(path, { expand, matchUrl, language: args.language }, DB, config);
  if (res.status >= 400) return Response.json({ success: false, error: res.statusText });

  return Response.json({
    success: true,
    endpoint_called: path,
    content_name: res.data?.name,
    content_id: res.data?.contentLink?.id ?? contentId,
    data: res.data,
    summary: 'Retrieved ' + res.data?.name + ' (ID: ' + contentId + ') with expand=' + expand
  });
}
Here is the execution handler from opti-baseline. It takes Mark's arguments, queries your CMS 12 Content Delivery API at /api/episerver/v3.0/content/{id}?expand=*, and returns the full JSON hierarchy back to Mark. It's read-only, cache-safe, and self-contained in Next.js.
